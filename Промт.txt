```python
# ─── ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ ДЛЯ ЗАДАНИЙ ──────────────────────────────────────

def fuzzy_match(user_answer: str, correct: str, threshold: float = 0.82) -> bool:
    """
    Нечёткое сравнение строк для обработки опечаток.
    Возвращает True если схожесть >= threshold.
    Также проверяет точное вхождение после нормализации.
    """
    ua = user_answer.strip().lower()
    ca = correct.strip().lower()
    
    # Точное совпадение
    if ua == ca:
        return True
    
    # Проверка содержания (короткий ответ внутри длинного)
    if len(ca) >= 3 and (ua in ca or ca in ua):
        return True
    
    # Нечёткое сравнение
    ratio = SequenceMatcher(None, ua, ca).ratio()
    return ratio >= threshold


def fuzzy_match_any(user_answer: str, correct_list: list, threshold: float = 0.82) -> bool:
    """Проверка на совпадение с любым вариантом из списка."""
    return any(fuzzy_match(user_answer, c, threshold) for c in correct_list)


def _hangman_display(word: str, guessed: set) -> str:
    """Отображение загадываемого слова с угаданными буквами."""
    result = []
    for ch in word.lower():
        if ch in guessed or ch == '-' or ch == ' ':
            result.append(ch.upper() if ch not in ('-', ' ') else ch)
        else:
            result.append('_')
    return ' '.join(result)


def _hangman_art(errors: int) -> str:
    """ASCII арт виселицы."""
    stages = [
        "```\n  +---+\n  |   |\n      |\n      |\n      |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n      |\n      |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n  |   |\n      |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n /|   |\n      |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n /|\\  |\n      |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n /|\\  |\n /    |\n      |\n=========```",
        "```\n  +---+\n  |   |\n  O   |\n /|\\  |\n / \\  |\n      |\n=========```",
    ]
    return stages[min(errors, len(stages) - 1)]
```

```python
async def job_task_timer(uid: int, cid: int, msg_id: int, deadline: float):
    """
    Ждёт до дедлайна и если задание не выполнено — отмечает провал.

    wait = max(0, deadline - time.time())
    await asyncio.sleep(wait + 1)

    session = job_sessions.pop(uid, None)

        return

            parse_mode="Markdown"
        )
        # Удаляем сообщение с заданием
        try:
            await bot.delete_message(cid, msg_id)
        except Exception:
            pass
    except Exception as e:
```

```python
# ─── ОБРАБОТКА ТЕКСТОВЫХ ОТВЕТОВ НА ЗАДАНИЯ ───────────────────────────────────

@dp.message()
async def handle_job_answer(m: types.Message):
    """
    Перехватывает текстовые сообщения и проверяет,
    является ли это ответом на активное задание.
    """
    try:
        uid = m.from_user.id
        cid = m.chat.id

        if uid not in job_sessions:
            return  # Не наш случай — пропускаем

        session = job_sessions.get(uid)
        if not session or session.get("chat_id") != cid:
            return  # Другой чат

        # Пропускаем команды
        if m.text and m.text.startswith("/"):
            return

        text = (m.text or "").strip()
        if not text:
            return

        now = time.time()

        # Проверяем дедлайн
        if now > session["deadline"]:
            job_sessions.pop(uid, None)
            return

        # Виселица — текстовые ответы не принимаем (только кнопки)
        if session["type"] == "hangman":
            return

        uname = session.get("uname", "Игрок")
        correct_answers = session["answers"]
        is_correct = fuzzy_match_any(text, correct_answers, threshold=0.82)

        if is_correct:
            # ── УСПЕХ ──────────────────────────────────────────────────────────
            job_sessions.pop(uid, None)
            await _job_success(uid, cid, session, m)
        else:
            # ── НЕВЕРНЫЙ ОТВЕТ ─────────────────────────────────────────────────
            session["attempts"] -= 1
            remaining = session["attempts"]

            if remaining <= 0:
                # Попытки исчерпаны
                job_sessions.pop(uid, None)
                await _job_fail(uid, cid, session, m, reason="attempts")
            else:
                time_left = session["deadline"] - now
                # Подсказка для флагов и географии
                hint = ""
                if session["type"] in ("geo_city_country", "geo_landmark"):
                    hint = f"\n💡 Напиши название страны полностью"
                elif session["type"] == "flag":
                    hint = f"\n💡 Напиши название страны на русском"
                elif session["type"] == "math":
                    hint = f"\n💡 Только число, без пробелов"

                try:
                    await m.reply(
                        f"❌ Неверно! Осталось попыток: {remaining}\n"
                        f"⏰ Времени: {format_time(time_left)}"
                        + hint,
                        parse_mode="Markdown"
                    )
                except Exception:
                    pass

    except Exception as e:
        logger.error(f"Job answer handler error: {e}", exc_info=True)


async def _job_success(uid: int, cid: int, session: dict, m: types.Message = None):
    """Начисляет награду и обновляет статистику после успешного задания."""
    try:
        u = await db_task(_get_user, uid, cid, session.get("uname", ""))
        uname = u.get("username") or session.get("uname", "Игрок")

        reward = session["reward"]
        reward_black = session.get("reward_black", 0)
        job_count = session.get("job_count", u.get("job_count", 0))
        new_job_count = job_count + 1

        # Базовая зарплата (как раньше) + бонус задания
        salary = calc_job_salary(new_job_count)
        base_earn = int(salary)
        bonus_salary = 1 if (new_job_count % 2 == 0 and salary != int(salary)) else 0
        total_coins = base_earn + bonus_salary + reward

        updates = {
            "money": u["money"] + total_coins,
            "last_job": session["now"],
            "job_count": new_job_count,
        }
        if reward_black:
            updates["black_money"] = u["black_money"] + reward_black

        await async_upd(uid, cid, updates)

        diff_labels = {"easy": "🟢 Лёгкое", "medium": "🟡 Среднее", "hard": "🔴 Сложное", "extreme": "🟣 Экстремальное"}
        diff = session.get("difficulty", "easy")

        black_str = f" +{reward_black}🖤" if reward_black else ""
        promo_msg = ""
        old_salary = calc_job_salary(job_count)
        new_salary_val = calc_job_salary(new_job_count)
        if new_salary_val > old_salary:
            promo_msg = f"\n🎉 Повышение! Зарплата: {new_salary_val}💰/раз"

        text = (
            f"✅ {uname} выполнил задание!\n"
            f"{diff_labels.get(diff, diff)}\n"
            f"Правильный ответ: **{session['answer']}**\n\n"
            f"💰 Зарплата: +{base_earn + bonus_salary}💰\n"
            f"🎁 Бонус за задание: +{reward}💰{black_str}\n"
            f"💼 Итого: +{total_coins}💰{black_str}\n"
            f"💰 Баланс: {u['money'] + total_coins}💰"
            + promo_msg
        )

        if m:
            await m.reply(text, parse_mode="Markdown")
        else:
            await bot.send_message(cid, text, parse_mode="Markdown")

        # Удаляем сообщение с заданием
        try:
            await bot.delete_message(cid, session["msg_id"])
        except Exception:
            pass

    except Exception as e:
        logger.error(f"Job success error: {e}", exc_info=True)


async def _job_fail(uid: int, cid: int, session: dict, m: types.Message = None, reason: str = "attempts"):
    """Обрабатывает провал задания."""
    try:
        uname = session.get("uname", "Игрок")

        if reason == "attempts":
            fail_text = f"💔 {uname}, попытки исчерпаны!"
        elif reason == "give_up":
            fail_text = f"🏳️ {uname} сдался!"
        else:
            fail_text = f"⏰ {uname}, время вышло!"

        text = (
            f"{fail_text}\n"
            f"Правильный ответ: **{session['answer']}**\n"
            f"💼 Зарплата не начислена.\n"
            f"📝 Кулдаун работы НЕ обновлён — попробуй снова!"
        )

        if m:
            await m.reply(text, parse_mode="Markdown")
        else:
            await bot.send_message(cid, text, parse_mode="Markdown")

        # Удаляем сообщение с заданием
        try:
            await bot.delete_message(cid, session["msg_id"])
        except Exception:
            pass

    except Exception as e:
        logger.error(f"Job fail error: {e}", exc_info=True)

@dp.callback_query(lambda c: c.data.startswith("job_give_up:"))

@dp.callback_query(lambda c: c.data.startswith("job_flag:"))
async def job_flag_cb(call: types.CallbackQuery):
    try:
        parts = call.data.split(":")
        target_uid = int(parts[1])
        chosen_flag = parts[2]

        if call.from_user.id != target_uid:
            return await call.answer("❌ Это не ваше задание!", show_alert=True)

        uid = call.from_user.id
        cid = call.message.chat.id
        now = time.time()

        session = job_sessions.get(uid)
        if not session:
            return await call.answer("❌ Нет активного задания!", show_alert=True)

        if now > session["deadline"]:
            job_sessions.pop(uid, None)
            await call.answer("⏰ Время вышло!", show_alert=True)
            await _job_fail(uid, cid, session, reason="timeout")
            return

        correct_flag = session["extra"]["correct_flag"]

        if chosen_flag == correct_flag:
            job_sessions.pop(uid, None)
            await call.answer("✅ Правильно!", show_alert=False)
            # Имитируем Message для _job_success
            await _job_success(uid, cid, session)
            try:
                await call.message.delete()
            except Exception:
                pass
        else:
            job_sessions.pop(uid, None)
            # Находим правильное название страны
            correct_country = FLAGS.get(correct_flag, ["?"])[0]
            await call.answer(f"❌ Неверно! Это был флаг {correct_country}", show_alert=True)
            await _job_fail(uid, cid, session, reason="attempts")

    except Exception as e:
        logger.error(f"Flag cb error: {e}", exc_info=True)
        await call.answer("⚠️ Ошибка", show_alert=True)


# ─── CALLBACK: Виселица ────────────────────────────────────────────────────────

@dp.callback_query(lambda c: c.data.startswith("hm:"))
async def hangman_cb(call: types.CallbackQuery):
    try:
        parts = call.data.split(":")
        target_uid = int(parts[1])
        letter = parts[2].lower()

        if call.from_user.id != target_uid:
            return await call.answer("❌ Это не ваша игра!", show_alert=True)

        uid = call.from_user.id
        cid = call.message.chat.id
        now = time.time()

        session = job_sessions.get(uid)
        if not session or session["type"] != "hangman":
            return await call.answer("❌ Нет активной игры!", show_alert=True)

        extra = session["extra"]
        word = extra["word"]
        guessed: set = extra["guessed"]
        errors: int = extra["errors"]
        max_errors: int = extra["max_errors"]

        if letter in guessed:
            return await call.answer("Эта буква уже была!", show_alert=False)

        guessed.add(letter)

        if letter not in word.lower():
            errors += 1
            extra["errors"] = errors

        extra["guessed"] = guessed

        display = _hangman_display(word, guessed)
        art = _hangman_art(errors)

        # Проверяем победу (все буквы слова угаданы)
        all_guessed = all(ch in guessed for ch in word.lower() if ch.isalpha())

        diff = session.get("difficulty", "medium")
        diff_labels = {"easy": "🟢 Лёгкое", "medium": "🟡 Среднее", "hard": "🔴 Сложное"}
        reward = session["reward"]
        reward_black = session.get("reward_black", 0)
        reward_txt = f"+{reward}💰" + (f" +{reward_black}🖤" if reward_black else "")
        uname = session.get("uname", "Игрок")

        if all_guessed:
            # ПОБЕДА
            job_sessions.pop(uid, None)
            await call.answer("🎉 СЛОВО УГАДАНО!", show_alert=True)
            text = (
                f"💼 {uname}, задание по работе!\n"
                f"{diff_labels.get(diff, diff)} | Награда: {reward_txt}\n\n"
                f"{extra['hint']}\n\n"
                f"✅ ПОБЕДА!\n"
                f"🔤 Слово: `{word.upper()}`\n"
                f"❌ Ошибки: {errors}/{max_errors}"
            )
            try:
                await call.message.edit_text(text, parse_mode="Markdown", reply_markup=None)
            except Exception:
                pass
            await _job_success(uid, cid, session)

        elif errors >= max_errors:
            # ПРОИГРЫШ
            job_sessions.pop(uid, None)
            await call.answer(f"💀 Проигрыш! Слово: {word.upper()}", show_alert=True)
            text = (
                f"💼 {uname}, задание по работе!\n"
                f"{diff_labels.get(diff, diff)} | Награда: {reward_txt}\n\n"
                f"{extra['hint']}\n\n"
                f"{art}\n\n"
                f"💀 ПРОИГРЫШ!\n"
                f"🔤 Слово было: `{word.upper()}`"
            )
            try:
                await call.message.edit_text(text, parse_mode="Markdown", reply_markup=None)
            except Exception:
                pass
            await _job_fail(uid, cid, session, reason="attempts")

        else:
            # Обновляем клавиатуру и текст
            await call.answer(
                f"{'✅' if letter in word.lower() else '❌'} Буква {letter.upper()}",
                show_alert=False
            )
            kb = _make_hangman_keyboard(uid, word, guessed, errors)
            text = (
                f"💼 {uname}, задание по работе!\n"
                f"{diff_labels.get(diff, diff)} | Награда: {reward_txt}\n\n"
                f"{extra['hint']}\n\n"
                f"{art}\n\n"
                f"🔤 Слово: `{display}`\n"
                f"❌ Ошибки: {errors}/{max_errors}\n"
                f"Угаданные буквы: {', '.join(sorted(guessed)) or '—'}"
            )
            try:
                await call.message.edit_text(text, reply_markup=kb, parse_mode="Markdown")
            except Exception:
                pass

    except Exception as e:
        logger.error(f"Hangman cb error: {e}", exc_info=True)
        await call.answer("⚠️ Ошибка", show_alert=True)
```

| 🏛️ Назови столицу | По стране назвать столицу | Сложная | +7💰+1🖤 |

**Ключевые особенности:**
- **Нечёткое сравнение 82%** — «Белорусь», «Белоуссия» → засчитается как «Беларусь»
- **Динамический таймер** — 30-120 сек в зависимости от сложности
- **3 попытки** для текстовых заданий, **6 ошибок** для виселицы
- **При провале — кулдаун НЕ обновляется**, игрок может попробовать сразу
- **500+ уникальных вопросов** в базе данных
- Виселица с **кнопками-буквами** и ASCII артом виселицы
- Флаги — **инлайн-выбор из 4 вариантов** без ввода текста
